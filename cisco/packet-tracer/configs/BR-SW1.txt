! =====================================================================
! BR-SW1  -  Branch access switch: VLANs, trunk, port security, DHCP snooping
! Target: Cisco IOSvL2 (G0/0 trunk; G0/1-2 access). Adapt port names.
! =====================================================================
enable
configure terminal
hostname BR-SW1
no ip domain-lookup
ip domain-name corp.lab
service password-encryption
service timestamps log datetime msec
enable secret <ENABLE_SECRET>
username netadmin privilege 15 secret <NETADMIN_SECRET>
banner motd ^ Authorized access only - CORP lab device BR-SW1 ^
!
vlan 30
 name HR
vlan 40
 name FINANCE
vlan 199
 name MANAGEMENT
vlan 999
 name PARKING-UNUSED
!
spanning-tree mode rapid-pvst
spanning-tree portfast bpduguard default
!
ip dhcp snooping vlan 30,40
no ip dhcp snooping information option
ip dhcp snooping
!
interface GigabitEthernet0/1
 description TRUNK-TO-BR-R2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,40,199
 switchport nonegotiate
 ip dhcp snooping trust
 no shutdown
!
interface FastEthernet0/1
 description HR-PC
 switchport mode access
 switchport access vlan 30
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 10
!
interface FastEthernet0/2
 description FINANCE-PC
 switchport mode access
 switchport access vlan 40
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 10
!
interface range FastEthernet0/3 - 24, GigabitEthernet0/2
 description UNUSED
 switchport mode access
 switchport access vlan 999
 shutdown
!
interface Vlan199
 description MANAGEMENT-SVI
 ip address 192.168.199.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.199.1
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line con 0
 login local
 logging synchronous
line vty 0 4
 login local
 transport input ssh
 exec-timeout 15 0
!
logging host 192.168.50.10
end
write memory
