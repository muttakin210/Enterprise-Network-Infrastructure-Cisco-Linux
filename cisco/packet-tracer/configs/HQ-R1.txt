! =====================================================================
! HQ-R1  -  HQ router: router-on-a-stick gateways, WAN, OSPF area 0
! Target: Packet Tracer ISR 2911 (G0/0 = 802.1Q trunk to HQ-SW1, G0/1 = WAN)
! Sanitized: replace every <PLACEHOLDER> before use. Never commit secrets.
! =====================================================================
enable
configure terminal
hostname HQ-R1
no ip domain-lookup
ip domain-name corp.lab
service password-encryption
service timestamps log datetime msec
service timestamps debug datetime msec
enable secret <ENABLE_SECRET>
username netadmin privilege 15 secret <NETADMIN_SECRET>
login block-for 120 attempts 5 within 60
banner motd ^ Authorized access only - CORP lab device HQ-R1 ^
!
! ---------- LAN: 802.1Q subinterfaces (gateways) ----------
interface GigabitEthernet0/0
 description TRUNK-TO-HQ-SW1
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.10
 description VLAN10-ADMIN
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip helper-address 192.168.50.10
!
interface GigabitEthernet0/0.20
 description VLAN20-IT
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip helper-address 192.168.50.10
!
interface GigabitEthernet0/0.50
 description VLAN50-SERVERS
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.0
!
interface GigabitEthernet0/0.99
 description VLAN99-MANAGEMENT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
 ip access-group MGMT-PROTECT out
!
! ---------- WAN ----------
interface GigabitEthernet0/1
 description WAN-to-BR-R2
 ip address 10.0.12.1 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
! ---------- OSPF ----------
router ospf 1
 router-id 1.1.1.1
 passive-interface default
 no passive-interface GigabitEthernet0/1
 network 10.0.12.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.50.0 0.0.0.255 area 0
 network 192.168.99.0 0.0.0.255 area 0
!
! ---------- Access policy ----------
! VLAN 99 is reachable only from the Admin VLAN, the server and the branch
! management VLAN. Everything else is dropped and logged.
ip access-list extended MGMT-PROTECT
 remark Who may reach VLAN 99 (switch management)
 permit ip 192.168.10.0 0.0.0.255 192.168.99.0 0.0.0.255
 permit ip host 192.168.50.10 192.168.99.0 0.0.0.255
 permit ip 192.168.199.0 0.0.0.255 192.168.99.0 0.0.0.255
 deny   ip any any
!
ip access-list standard VTY-ADMIN
 remark Who may open SSH to this router
 permit 192.168.10.0 0.0.0.255
 permit 192.168.99.0 0.0.0.255
 permit 192.168.199.0 0.0.0.255
 permit host 192.168.50.10
 deny   any
!
! ---------- SSH v2 only ----------
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
line con 0
 login local
 logging synchronous
 exec-timeout 15 0
line vty 0 4
 login local
 transport input ssh
 access-class VTY-ADMIN in
 exec-timeout 15 0
!
! ---------- Syslog ----------
logging host 192.168.50.10
!
! ---------- SNMPv3 (authPriv) ----------
!
ntp server 192.168.50.10
end
write memory
