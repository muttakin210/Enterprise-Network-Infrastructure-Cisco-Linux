! =====================================================================
! HQ-SW1  -  HQ access switch: VLANs, trunk, port security, DHCP snooping
! Target: Packet Tracer 2960-24TT (Gi0/1 trunk; Fa0/x access).
! =====================================================================
enable
configure terminal
hostname HQ-SW1
no ip domain-lookup
ip domain-name corp.lab
service password-encryption
service timestamps log datetime msec
enable secret <ENABLE_SECRET>
username netadmin privilege 15 secret <NETADMIN_SECRET>
banner motd ^ Authorized access only - CORP lab device HQ-SW1 ^
!
vlan 10
 name ADMIN
vlan 20
 name IT
vlan 50
 name SERVERS
vlan 99
 name MANAGEMENT
vlan 999
 name PARKING-UNUSED
!
spanning-tree mode rapid-pvst
spanning-tree portfast bpduguard default
!
! ---------- DHCP snooping (only the uplink is trusted) ----------
ip dhcp snooping vlan 10,20
no ip dhcp snooping information option
ip dhcp snooping
!
interface GigabitEthernet0/1
 description TRUNK-TO-HQ-R1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,50,99
 switchport nonegotiate
 ip dhcp snooping trust
 no shutdown
!
interface FastEthernet0/1
 description ADMIN-PC
 switchport mode access
 switchport access vlan 10
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 10
!
interface FastEthernet0/2
 description IT-PC
 switchport mode access
 switchport access vlan 20
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 10
!
interface FastEthernet0/3
 description UBUNTU-SERVER
 switchport mode access
 switchport access vlan 50
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
!
! Unused ports: shut and park (adapt the range to the image)
interface range FastEthernet0/4 - 24, GigabitEthernet0/2
 description UNUSED
 switchport mode access
 switchport access vlan 999
 shutdown
!
interface Vlan99
 description MANAGEMENT-SVI
 ip address 192.168.99.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line con 0
 login local
 logging synchronous
line vty 0 4
 login local
 transport input ssh
 exec-timeout 15 0
!
logging host 192.168.50.10
logging trap informational
end
write memory
