! =====================================================================
! BR-R2  -  Branch router: router-on-a-stick gateways, WAN, OSPF area 0
! Target: Packet Tracer ISR 2911 (G0/0 = 802.1Q trunk to BR-SW1, G0/1 = WAN)
! Sanitized: replace every <PLACEHOLDER> before use. Never commit secrets.
! =====================================================================
enable
configure terminal
hostname BR-R2
no ip domain-lookup
ip domain-name corp.lab
service password-encryption
service timestamps log datetime msec
service timestamps debug datetime msec
enable secret <ENABLE_SECRET>
username netadmin privilege 15 secret <NETADMIN_SECRET>
login block-for 120 attempts 5 within 60
banner motd ^ Authorized access only - CORP lab device BR-R2 ^
!
interface GigabitEthernet0/0
 description TRUNK-TO-BR-SW1
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.30
 description VLAN30-HR
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip helper-address 192.168.50.10
!
interface GigabitEthernet0/0.40
 description VLAN40-FINANCE
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip helper-address 192.168.50.10
 ip access-group FINANCE-PROTECT out
!
interface GigabitEthernet0/0.199
 description VLAN199-MANAGEMENT
 encapsulation dot1Q 199
 ip address 192.168.199.1 255.255.255.0
 ip access-group MGMT-PROTECT out
!
interface GigabitEthernet0/1
 description WAN-to-HQ-R1
 ip address 10.0.12.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
router ospf 1
 router-id 2.2.2.2
 passive-interface default
 no passive-interface GigabitEthernet0/1
 network 10.0.12.0 0.0.0.3 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.40.0 0.0.0.255 area 0
 network 192.168.199.0 0.0.0.255 area 0
!
! Finance VLAN: IT (20) and HR (30) may not reach Finance hosts.
! The Admin VLAN and the server (DNS/DHCP/web) are allowed.
ip access-list extended FINANCE-PROTECT
 remark Traffic routed INTO VLAN 40
 permit ip 192.168.10.0 0.0.0.255 192.168.40.0 0.0.0.255
 permit ip host 192.168.50.10 192.168.40.0 0.0.0.255
 deny   ip 192.168.20.0 0.0.0.255 192.168.40.0 0.0.0.255
 deny   ip 192.168.30.0 0.0.0.255 192.168.40.0 0.0.0.255
 permit ip any 192.168.40.0 0.0.0.255
!
ip access-list extended MGMT-PROTECT
 remark Who may reach VLAN 199 (switch management)
 permit ip 192.168.10.0 0.0.0.255 192.168.199.0 0.0.0.255
 permit ip host 192.168.50.10 192.168.199.0 0.0.0.255
 permit ip 192.168.99.0 0.0.0.255 192.168.199.0 0.0.0.255
 deny   ip any any
!
ip access-list standard VTY-ADMIN
 permit 192.168.10.0 0.0.0.255
 permit 192.168.99.0 0.0.0.255
 permit 192.168.199.0 0.0.0.255
 permit host 192.168.50.10
 deny   any
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
line con 0
 login local
 logging synchronous
 exec-timeout 15 0
line vty 0 4
 login local
 transport input ssh
 access-class VTY-ADMIN in
 exec-timeout 15 0
!
logging host 192.168.50.10
!
!
ntp server 192.168.50.10
end
write memory
